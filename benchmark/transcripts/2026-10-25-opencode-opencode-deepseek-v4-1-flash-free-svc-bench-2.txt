import cv2, numpy as np, json
cap=cv2.VideoCapture("dribble-count.mp4")
n=int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
rows=[]
for fi in range(n):
    ok,img=cap.read()
    if not ok: break
    b,g,r=cv2.split(img.astype(int))
    ball=((r>195)&(g>105)&(b<95)).astype(np.uint8)
    blue=((b>150)&(r<120)&(g<140)).astype(np.uint8)
    green=((g>140)&(r<110)&(b<110)).astype(np.uint8)
    def cents(m,minarea=20):
        nn,lab,st,ce=cv2.connectedComponentsWithStats(m,8)
        return [(float(ce[i,0]),float(ce[i,1]),int(st[i,4])) for i in range(1,nn) if st[i,4]>=minarea]
    bl=cents(ball,15)
    # merge blobs within 18 px
    merged=[]
    for c in bl:
        for m in merged:
            if abs(c[0]-m[0])<18 and abs(c[1]-m[1])<18:
                m[0]=(m[0]*m[2]+c[0]*c[2])/(m[2]+c[2]); m[1]=(m[1]*m[2]+c[1]*c[2])/(m[2]+c[2]); m[2]+=c[2]
                break
        else:
            merged.append([c[0],c[1],c[2]])
    bl=cent_b=[(m[0],m[1],m[2]) for m in merged]
    bpl=cents(blue,200); gpl=cents(green,200)
    # main player comp largest
    bx = max(bpl,key=lambda t:t[2]) if bpl else None
    gx = max(gpl,key=lambda t:t[2]) if gpl else None
    white=((r>200)&(g>200)&(b>200)).astype(np.uint8)
    wr=white.sum(axis=1)
    ys=[y for y in range(len(wr)) if wr[y]>300]
    liney = float(np.mean(ys)) if ys else None
    rows.append(dict(f=fi, balls=bl, blue=bx, green=gx, line=liney))
json.dump(rows,open("work/dribble_raw.json","w"))
print(n, rows[0], rows[100])
